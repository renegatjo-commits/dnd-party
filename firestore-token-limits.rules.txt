// В действующих правилах Firebase Firestore замените только блок
// match /tokens/{tokenId} { ... } на блок ниже.
// Остальные разделы (game, characters, spells, overlays, Phasmo, ярмарка)
// сохраняются. Функции isMaster() и isPlayer() уже есть в ваших правилах.

match /tokens/{tokenId} {
  function character() {
    return get(/databases/$(database)/documents/characters/$(tokenId)).data;
  }
  function game() {
    return get(/databases/$(database)/documents/game/main).data;
  }
  function insideTurnRadius() {
    let g = game();
    let c = character();
    let origins = g.get('turnOrigins', {});
    let p = origins.get(tokenId, {});
    let d = request.resource.data;
    let speed = c.get('speed', 0);
    let cell = g.get('cellSize', 64);
    let dx = d.x - p.get('x', -999999);
    let dy = d.y - p.get('y', -999999);
    let radius = speed * cell;
    return g.get('currentTurn', 0) is int
      && g.get('currentTurn', 0) >= 1
      && origins is map && p is map
      && p.keys().hasAll(['x', 'y'])
      && p.x is number && p.y is number
      && speed is number && speed >= 0
      && cell is number && cell > 0
      && d.x is number && d.y is number
      && d.x >= 0 && d.x <= g.get('mapWidth', 1280)
      && d.y >= 0 && d.y <= g.get('mapHeight', 800)
      && dx * dx + dy * dy <= radius * radius + 0.0001;
  }
  allow read: if isPlayer();
  allow create, delete: if isMaster();
  allow update: if isMaster();
  allow update: if isPlayer()
    && resource.data.get('npc', false) == false
    && character().get('isNpc', false) == false
    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['x', 'y'])
    && insideTurnRadius();
}
